# 2. Пересобираем бэкенд без кэша
docker build --no-cache -t n8n-backend:latest ./backend

# 3. Пересобираем фронтенд (теперь он намертво вшьет правильные ключи из .env)
docker build --no-cache -t n8n-frontend:latest ./frontend

# 4. Перезапускаем все сервисы, чтобы применить новые образы
docker compose up -d --force-recreate
