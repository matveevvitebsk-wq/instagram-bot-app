# Перейти в каталог проекта
cd /opt/instagram-bot/instagram-bot-app

# Если есть локальные изменения в package-файлах
git checkout -- frontend/package.json frontend/package-lock.json

# Получить последние изменения
git pull origin main

# Полностью остановить проект
docker compose down

# Пересобрать backend без кэша
docker build --no-cache -t instagram-bot-app-backend ./backend

# Пересобрать frontend без кэша
docker build --no-cache -t instagram-bot-app-frontend ./frontend

# Пересоздать все контейнеры
docker compose up -d --force-recreate

# Проверить состояние контейнеров
docker compose ps

# Если backend не запустился — посмотреть последние 100 строк логов
docker logs backend --tail 100

# Проверить конфигурацию nginx
docker exec nginx nginx -t

# Проверить использование процессора
top -b -n 1 -o %CPU | head -n 20
